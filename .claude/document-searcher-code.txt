#### `build.rs` (Compiles Blueprint & GResource & GSchema)
```rust
use std::env;
use std::path::PathBuf;
use std::process::Command;

fn main() {
    // 1. Compile Blueprint (.blp) -> .xml
    let out_dir = PathBuf::from(env::var("OUT_DIR").unwrap());
    let resource_dir = PathBuf::from("data/resources");
    
    // Ensure output directory exists
    std::fs::create_dir_all(&out_dir).unwrap();

    // Compile window.blp -> window.xml
    let blp_input = resource_dir.join("window.blp");
    let xml_output = out_dir.join("window.xml");
    
    let status = Command::new("blueprint-compiler")
        .args(["compile", "--output", xml_output.to_str().unwrap(), blp_input.to_str().unwrap()])
        .status()
        .expect("Failed to run blueprint-compiler. Install blueprint-compiler (ALT: blueprint-compiler package).");
    
    if !status.success() {
        panic!("Blueprint compilation failed");
    }

    // 2. Compile GSchema (if exists)
    let gschema_input = resource_dir.join("org.gtk.DocumentSearcher.gschema.xml");
    if gschema_input.exists() {
        let schema_out_dir = out_dir.join("schemas");
        std::fs::create_dir_all(&schema_out_dir).unwrap();
        Command::new("glib-compile-schemas")
            .arg("--targetdir")
            .arg(&schema_out_dir)
            .arg(&resource_dir)
            .status()
            .expect("Failed to compile schemas");
        // Tell cargo to link the compiled schemas
        println!("cargo:rustc-env=GSETTINGS_SCHEMA_DIR={}", schema_out_dir.display());
    }

    // 3. Compile GResource (.gresource.xml -> binary)
    // We generate gresource.xml dynamically to include the compiled window.xml
    let gresource_xml = format!(r#"
<?xml version="1.0" encoding="UTF-8"?>
<gresources>
  <gresource prefix="/org/gtk/DocumentSearcher">
    <file>window.xml</file>
  </gresource>
</gresources>
"#);
    let gresource_xml_path = out_dir.join("document-searcher.gresource.xml");
    std::fs::write(&gresource_xml_path, gresource_xml).unwrap();

    gresource::compile(&gresource_xml_path, &out_dir).unwrap();

    // 4. Tell Cargo to re-run if UI files change
    println!("cargo:rerun-if-changed=data/resources/window.blp");
    println!("cargo:rerun-if-changed=data/resources/gresource.xml"); 
    println!("cargo:rerun-if-changed=data/resources/org.gtk.DocumentSearcher.gschema.xml");
    
    // 5. Include the compiled .gresource binary in the binary
    // The binary is at OUT_DIR/document-searcher.gresource
    // We use `gresource` crate macro in main.rs to load it, but we need the path.
    // Actually, `gresource::compile` outputs to OUT_DIR. We can use `include_bytes` in main.rs 
    // pointing to the output file, but it's easier to use the `gresource` crate's `simple!` macro 
    // or just load from path at runtime. 
    // Standard Relm4 pattern: compile to OUT_DIR, load via `gio::Resource::load` in main.
    println!("cargo:rustc-env=GRESOURCE_PATH={}", out_dir.join("document-searcher.gresource").display());
}
```

```rust
use relm4::{RelmApp, RelmAppExt};
use std::env;

mod app;
mod model;
mod utils;

fn main() {
    // Initialize tracing for logs (visible in flatpak logs / journalctl)
    tracing_subscriber::fmt()
        .with_env_filter(tracing_subscriber::EnvFilter::from_default_env())
        .init();

    // Load GSettings Schema (compiled by build.rs)
    // We must set the schema dir before glib initializes settings usually, 
    // but RelmApp runs glib::init internally. 
    // We rely on GSETTINGS_SCHEMA_DIR env var set by build.rs or install prefix.
    // For Flatpak, schemas are installed to /app/share/glib-2.0/schemas.
    // For dev, build.rs sets GSETTINGS_SCHEMA_DIR.

    // Load GResource
    let res_path = env::var("GRESOURCE_PATH").unwrap_or_else(|_| {
        // Fallback for installed location (Flatpak)
        "/app/lib/org.gtk.DocumentSearcher/document-searcher.gresource".to_string()
    });
    
    let resource = gio::Resource::load(&res_path).expect(&format!("Failed to load gresource from: {}", res_path));
    gio::resources_register(&resource);

    // Run Application
    let app = RelmApp::new("org.gtk.DocumentSearcher");
    app.run::<app::AppComponent>(());
}
```

#### `src/model/extractor.rs` (Text Extraction Logic)
```rust
use anyhow::{Context, Result};
use std::path::Path;
use lopdf::Document as PdfDoc;
use docx_rs::read_docx;
use pdf_extract::extract_text;
use leptess::{LepTess, Variable};
use mime_guess::from_path;
use unicode_segmentation::UnicodeSegmentation;

pub trait TextExtractor {
    fn extract(&self, path: &Path) -> Result<String>;
}

pub struct CompositeExtractor;

impl CompositeExtractor {
    pub fn new() -> Self {
        Self
    }
}

impl TextExtractor for CompositeExtractor {
    fn extract(&self, path: &Path) -> Result<String> {
        let mime = from_path(path).first_or_octet_stream();
        
        match (mime.type_(), mime.subtype()) {
            ("text", "plain") => self.extract_txt(path),
            ("application", "vnd.openxmlformats-officedocument.wordprocessingml.document") => self.extract_docx(path),
            ("application", "pdf") => self.extract_pdf(path),
            ("image", _) => self.extract_ocr(path), // OCR for images
            _ => {
                // Fallback: try PDF extraction (some PDFs misidentified), then OCR
                self.extract_pdf(path).or_else(|_| self.extract_ocr(path))
            }
        }
    }
}

impl CompositeExtractor {
    fn extract_txt(&self, path: &Path) -> Result<String> {
        std::fs::read_to_string(path).context("Reading txt file")
    }

    fn extract_docx(&self, path: &Path) -> Result<String> {
        let file = std::fs::File::open(path).context("Opening docx")?;
        let docx = read_docx(file).context("Parsing docx")?;
        // docx_rs returns a Docx struct. We need to iterate paragraphs.
        let mut text = String::new();
        for para in docx.paragraphs {
            for run in para.runs {
                if let Some(t) = run.text {
                    text.push_str(&t);
                }
            }
            text.push('
');
        }
        Ok(text)
    }

    fn extract_pdf(&self, path: &Path) -> Result<String> {
        // Try pdf-extract first (fast, native text)
        if let Ok(text) = extract_text(path) {
            if !text.trim().is_empty() {
                return Ok(text);
            }
        }
        // Fallback to lopdf for more control or if pdf-extract fails
        let doc = PdfDoc::load(path).context("Loading PDF with lopdf")?;
        let mut text = String::new();
        for page_num in 1..=doc.get_pages().len() {
            if let Ok(page_text) = doc.extract_text(&[page_num as u32]) {
                text.push_str(&page_text);
                text.push('
');
            }
        }
        if text.trim().is_empty() {
            anyhow::bail!("No text layer found in PDF");
        }
        Ok(text)
    }

    fn extract_ocr(&self, path: &Path) -> Result<String> {
        // Requires tesseract installed in Flatpak (org.freedesktop.Platform.ffmpeg-full or tesseract extension)
        // Language: "eng" default. Could be configurable.
        let mut lt = LepTess::new(None, "eng").context("Init Tesseract")?;
        lt.set_image_path(path.to_str().unwrap()).context("Set image for OCR")?;
        let text = lt.get_utf8_text().context("OCR Recognition")?;
        Ok(text)
    }
}

/// Finds all occurrences of `needle` in `haystack` (case-insensitive) 
/// and returns context (20 chars before/after).
pub fn find_occurrences(haystack: &str, needle: &str) -> Vec<SearchOccurrence> {
    if needle.is_empty() {
        return vec![];
    }
    
    let haystack_lower = haystack.to_lowercase();
    let needle_lower = needle.to_lowercase();
    let mut occurrences = Vec::new();
    let mut start = 0;

    // Use grapheme clusters for correct char counting (unicode-segmentation)
    let graphemes: Vec<&str> = haystack.graphemes(true).collect();
    let graphemes_lower: Vec<String> = haystack_lower.graphemes(true).map(String::from).collect();
    let needle_graphemes: Vec<String> = needle_lower.graphemes(true).map(String::from).collect();
    let needle_len = needle_graphemes.len();

    if needle_len == 0 { return vec![]; }

    while start <= graphemes_lower.len().saturating_sub(needle_len) {
        let window = &graphemes_lower[start..start + needle_len];
        if window == needle_graphemes {
            // Match found at grapheme index `start`
            // Calculate byte indices for slicing original string
            let byte_start: usize = graphemes[..start].concat().len();
            let byte_end: usize = graphemes[..start + needle_len].concat().len();
            
            // Context: 20 graphemes before/after
            let ctx_before_start = start.saturating_sub(20);
            let ctx_before_bytes: usize = graphemes[..ctx_before_start].concat().len();
            let ctx_after_end = (start + needle_len + 20).min(graphemes.len());
            let ctx_after_bytes: usize = graphemes[..ctx_after_end].concat().len();

            let context_before = &haystack[ctx_before_bytes..byte_start];
            let match_text = &haystack[byte_start..byte_end];
            let context_after = &haystack[byte_end..ctx_after_bytes];

            occurrences.push(SearchOccurrence {
                context_before: context_before.to_string(),
                match_text: match_text.to_string(),
                context_after: context_after.to_string(),
            });

            start += needle_len; // Move past match
        } else {
            start += 1;
        }
    }
    occurrences
}
```

```rust
use crate::model::{extractor::{CompositeExtractor, TextExtractor, find_occurrences}, msg::WorkerMsg, state::FoundDocument};
use anyhow::Result;
use async_channel::{Receiver, Sender};
use std::path::PathBuf;
use tokio::fs;
use walkdir::WalkDir;

pub async fn run_worker(
    receiver: Receiver<WorkerMsg>,
    ui_sender: Sender<crate::model::msg::AppMsg>,
) {
    let extractor = CompositeExtractor::new();

    while let Ok(msg) = receiver.recv().await {
        match msg {
            WorkerMsg::ScanFolder(root) => {
                let _ = ui_sender.send(crate::model::msg::AppMsg::ScanStarted).await;
                let mut docs = Vec::new();
                
                // Walkdir is blocking, use spawn_blocking
                let root_clone = root.clone();
                let result = tokio::task::spawn_blocking(move || {
                    let mut local_docs = Vec::new();
                    for entry in WalkDir::new(&root_clone).follow_links(true).into_iter().filter_map(|e| e.ok()) {
                        if entry.file_type().is_file() {
                            let abs_path = entry.path().to_path_buf();
                            // Filter by extension early to avoid processing binary junk
                            if is_supported_document(&abs_path) {
                                if let Ok(rel_path) = abs_path.strip_prefix(&root_clone) {
                                    local_docs.push(FoundDocument {
                                        relative_path: rel_path.to_string_lossy().to_string(),
                                        absolute_path: abs_path,
                                    });
                                }
                            }
                        }
                    }
                    local_docs
                }).await;

                match result {
                    Ok(d) => { let _ = ui_sender.send(crate::model::msg::AppMsg::ScanFinished(d)).await; }
                    Err(e) => { let _ = ui_sender.send(crate::model::msg::AppMsg::ScanError(e.to_string())).await; }
                }
            }

            WorkerMsg::SearchDocuments { query, documents } => {
                let _ = ui_sender.send(crate::model::msg::AppMsg::SearchStarted).await;
                let total = documents.len();
                
                for (idx, doc) in documents.into_iter().enumerate() {
                    // Send progress
                    let _ = ui_sender.send(crate::model::msg::AppMsg::SearchProgress(idx, total)).await;

                    // Extract text (blocking)
                    let abs_path = doc.absolute_path.clone();
                    let rel_path = doc.relative_path.clone();
                    let query_clone = query.clone();
                    
                    let text_result = tokio::task::spawn_blocking(move || {
                        extractor.extract(&abs_path)
                    }).await;

                    match text_result {
                        Ok(Ok(text)) => {
                            let occurrences = find_occurrences(&text, &query_clone);
                            if !occurrences.is_empty() {
                                let matched = crate::model::state::MatchedDocument {
                                    relative_path: rel_path,
                                    occurrences,
                                };
                                let _ = ui_sender.send(crate::model::msg::AppMsg::SearchResult(matched)).await;
                            }
                        }
                        Ok(Err(e)) => {
                            tracing::warn!("Failed to extract text from {}: {}", abs_path.display(), e);
                        }
                        Err(e) => {
                            tracing::error!("Task join error for {}: {}", abs_path.display(), e);
                        }
                    }
                }
                let _ = ui_sender.send(crate::model::msg::AppMsg::SearchFinished).await;
            }
        }
    }
}

fn is_supported_document(path: &PathBuf) -> bool {
    if let Some(ext) = path.extension().and_then(|s| s.to_str()) {
        matches!(ext.to_lowercase().as_str(), 
            "txt" | "md" | "rs" | "toml" | "json" | "xml" | "html" | "css" | "js" | "ts" | // Code/Text
            "pdf" | 
            "docx" | "doc" | // Word (doc not supported by docx-rs easily, skip or use antiword)
            "odt" | // LibreOffice (needs unzip + xml parsing)
            "rtf" |
            "png" | "jpg" | "jpeg" | "tiff" | "bmp" // Images for OCR
        )
    } else {
        false
    }
}
```

```rust
        // OR: Bind properties in Blueprint? Spinner visibility bound to model property? Blueprint doesn't bind arbitrary model fields.
        
        // BEST APPROACH for Relm4 0.7: Switch to `Component` trait (not SimpleComponent) which provides `update(&mut self, msg, sender, widgets)`.

    }

    fn update(&mut self, msg: Self::Input, sender: ComponentSender<Self>, widgets: &mut Self::Widgets) {
        match msg {
            AppMsg::LoadSettings => { /* ... */ 
                let settings = gio::Settings::new("org.gtk.DocumentSearcher");
                let path_str: String = settings.get("last-folder");
                let path = PathBuf::from(path_str);
                self.state.settings = AppSettings { last_folder: path.clone() };
                self.scan_folder(path);
            }
            AppMsg::ChangeFolderRequested => { /* ... spawn pick_folder ... */ }
            AppMsg::FolderSelected(Some(path)) => { 
                let settings = gio::Settings::new("org.gtk.DocumentSearcher");
                settings.set_string("last-folder", &path.to_string_lossy()).ok();
                self.state.settings.last_folder = path.clone();
                self.scan_folder(path);
            }
            AppMsg::FolderSelected(None) => {},
            
            AppMsg::ScanStarted => {
                self.state.found_documents.clear();
                self.files_factory.guard().clear();
                self.state.matched_documents.clear();
                self.results_factory.guard().clear();
                self.state.search_status = SearchStatus::Idle;
                self.sync_labels(widgets);
            }
            AppMsg::ScanFinished(docs) => {
                self.state.found_documents = docs;
                let mut guard = self.files_factory.guard();
                for doc in &self.state.found_documents {
                    guard.push_back(doc.relative_path.clone());
                }
                self.sync_labels(widgets);
            }
            AppMsg::ScanError(e) => {
                self.state.search_status = SearchStatus::Error(e.clone());
                self.show_toast(widgets, &format!("Scan failed: {}", e));
            }

            AppMsg::SearchQueryChanged(query) => {
                self.state.search_query = query.clone();
                if query.trim().is_empty() {
                    self.state.matched_documents.clear();
                    self.results_factory.guard().clear();
                    self.state.search_status = SearchStatus::Idle;
                    self.update_search_ui(widgets);
                    return;
                }
                self.start_search(query);
            }
            AppMsg::SearchStarted => {
                self.state.search_status = SearchStatus::Searching;
                self.state.matched_documents.clear();
                self.results_factory.guard().clear();
                self.update_search_ui(widgets);
            }
            AppMsg::SearchResult(matched) => {
                self.state.matched_documents.push(matched.clone());
                self.results_factory.guard().push_back(matched);
            }
            AppMsg::SearchFinished => {
                self.state.search_status = SearchStatus::Success;
                self.update_search_ui(widgets);
            }
            AppMsg::SearchError(e) => {
                self.state.search_status = SearchStatus::Error(e.clone());
                self.update_search_ui(widgets);
                self.show_toast(widgets, &format!("Search error: {}", e));
            }
            
            // Row Activation Handlers
            AppMsg::OpenFileListRow(idx) => { // New specific Msg
                if let Some(doc) = self.state.found_documents.get(idx as usize) {
                    open_in_nautilus(&doc.absolute_path);
                }
            }
            AppMsg::OpenResultRow(idx) => { // New specific Msg
                if let Some(doc) = self.state.matched_documents.get(idx as usize) {
                    // Find absolute path from found_documents
                    if let Some(found) = self.state.found_documents.iter().find(|d| d.relative_path == doc.relative_path) {
                        open_in_nautilus(&found.absolute_path);
                    }
                }
            }
        }
    }
```

### 6. Critical Implementation Details & Fixes

1.  **Msg Enum Update (`src/model/msg.rs`)**:
    Add the row activation variants:
    ```rust
    // In AppMsg enum
    OpenFileListRow(i32),
    OpenResultRow(i32),
    ```

2.  **Blueprint Signal Handlers**:
    In `window.blp`, the `search_entry` uses `connect_changed[sender] => ...`. This works with Relm4's `view!` macro automatically mapping `sender` to the component sender.

3.  **Tesseract in Flatpak**:
    The manifest includes a `tesseract-data` module copying host tessdata. **Better approach**: Use `org.freedesktop.Platform` extension for tesseract if available, or add `tesseract` as a module building from source. For ALT Linux Flatpak, usually `org.freedesktop.Platform` has tesseract libs but not data. You must bundle `tessdata` (like `eng.traineddata`) in your module sources.

4.  **OCR Dependency (`leptess`)**:
    Requires `liblept` and `libtesseract` dev files at compile time (provided by `org.freedesktop.Sdk`). At runtime, provided by `org.freedesktop.Platform` (check version compatibility).

5.  **Search Logic Performance**:
    The current `find_occurrences` uses `unicode-segmentation` for grapheme correctness. This is slower than byte search but correct for UI display. For large files, consider streaming search (not loading whole file into RAM), but `docx-rs`/`lopdf` load whole file anyway.

6.  **Thread Safety**:
    *   `WorkerMsg` sent via `async_channel` (MPSC).
    *   Worker runs `tokio::spawn_blocking` for CPU-heavy extraction.
    *   Results sent back to UI thread via `ComponentSender` (Relm4 handles `glib::MainContext` marshalling).

7.  **File Opening (Nautilus)**:
    `utils::open_in_nautilus` uses `gio::AppInfo::launch_default_for_uri("file://...")`. This works in Flatpak via `xdg-desktop-portal` (requires `--talk-name=org.freedesktop.portal.Desktop` which is implied by filechooser talk usually, but explicit is better).

8.  **Build Instructions**:
